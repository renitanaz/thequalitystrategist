Playwright: scan this page
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

test('playground has expected violations', async ({ page }) => {
  await page.goto('https://renitanaz.github.io/thequalitystrategist/blogs/accessibility/A11y-Test-Playground/2%20A11y%20Test%20Playground%20-%20Cost%20of%20Shift-Right.html');
  const results = await new AxeBuilder({ page })
    .withTags(['wcag2a', 'wcag2aa'])
    .analyze();
  // Should find: Critical (missing alt, missing label, empty button)
  //              Serious (colour contrast, invalid ARIA)
  //              Moderate (duplicate ID)
  console.log('Violations:', results.violations.length);
  console.log('Incomplete:', results.incomplete.length);
  console.log('*****************************************');

  results.violations.forEach(v => {
    console.log(`- [${v.impact}] ${v.id}: ${v.description}`);
  });

  // Since this playground is INTENTIONALLY broken, assert it finds violations
  expect(results.violations.length).toBeGreaterThan(0);
});
